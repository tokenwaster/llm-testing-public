<parameter name="content">import re
from collections import defaultdict

alias = {}
tx = {}

def acct(tok):
    tok = tok.strip()
    if tok.startswith('"'):
        return alias[tok.strip('"')]
    return tok

for line in open('ledger.txt', encoding='utf-8'):
    line = line.strip()
    m = re.match(r'^\[(.*?)\](.*)$', line)
    if not m:
        continue
    head, rest = m.group(1), m.group(2).strip()
    parts = head.split(None, 1)
    cmd = parts[0]
    if cmd == 'TXN':
        tid = parts[1]
        mm = re.match(r'^("[^"]+"|ACCT-\d+)\s+([+-]\d+)\s+(SETTLED|PENDING)$', rest)
        tx[tid] = {'acct': acct(mm.group(1)), 'amt': int(mm.group(2)), 'status': mm.group(3), 'void': False}
    elif cmd == 'VOID':
        tx[parts[1]]['void'] = True
    elif cmd == 'RESTORE':
        tx[parts[1]]['void'] = False
    elif cmd == 'AMEND':
        tid, amt = parts[1].split()
        tx[tid]['amt'] = int(amt)
    elif cmd == 'SETTLE':
        tx[parts[1]]['status'] = 'SETTLED'
    elif cmd == 'TRANSFER':
        tid, a = parts[1].split(None, 1)
        tx[tid]['acct'] = acct(a)
    elif cmd == 'VOID-ALL':
        a = acct(parts[1])
        for t in tx.values():
            if t['acct'] == a:
                t['void'] = True
    elif cmd == 'ALIAS':
        mm = re.match(r'^(ACCT-\d+)\s*=\s*"([^"]+)"$', parts[1])
        alias[mm.group(2)] = mm.group(1)
    else:
        raise Exception(line)

bal = defaultdict(int)
active = 0
for t in tx.values():
    if t['status'] == 'SETTLED' and not t['void']:
        bal[t['acct']] += t['amt']
        active += 1

accts = sorted(bal)
for a in accts:
    print(a, bal[a])
hi = max(accts, key=lambda a: (bal[a], -int(a[5:])))
lo = min(accts, key=lambda a: (bal[a], int(a[5:])))
print(f"HIGHEST_ACCOUNT: {hi}")
print(f"HIGHEST_BALANCE: {bal[hi]}")
print(f"LOWEST_ACCOUNT: {lo}")
print(f"NET_TOTAL: {sum(bal.values())}")
print(f"NUM_NEGATIVE: {sum(1 for a in accts if bal[a] < 0)}")
print(f"NUM_ACTIVE: {active}")
print(len(tx))
