def calculate_settled_balance(ledger):
    """Calculates the settled balance for each account."""
    balances = {}
    for entry in ledger:
        if entry["type"] == "TXN":
            account = entry["ACCT"]
            amount = float(entry["value"])
            if account not in balances:
                balances[account] = 0.0
            if "VOID" in entry:
                continue  # Skip voided transactions
            elif "AMEND" in entry:
                try:
                    amount = float(entry["value"].replace("+", "").replace("-", "")) # Handle amend amounts correctly
                except ValueError:
                    pass 
            balances[account] += amount
    return balances

def analyze_ledger(ledger):
    """Analyzes the ledger and calculates key metrics."""
    balances = calculate_settled_balance(ledger)
    
    highest_account = max(balances, key=lambda k: balances[k])
    highest_balance = balances[highest_account]
    
    lowest_account = min(balances, key=lambda k: balances[k])
    net_total = sum(balances.values())
    num_negative = sum(1 for balance in balances.values() if balance < 0)
    
    return highest_account, highest_balance, lowest_account, net_total, num_negative
