  final newline does not add an extra line).


- `top` is the most frequent word and its count. Ties are broken alphabetically


  (earliest wins). If the file has no words, print `top: - 0`.

```python


import sys


import re


from collections import Counter





def main():


    if len(sys.argv) != 2:


        return





    file_path = sys.argv[1]





    try:


        with open(file_path, 'r', encoding='utf-8') as f:


            content = f.read()


    except FileNotFoundError:


        return


    except Exception:


        return





    # Count lines


    if not content:


        num_lines = 0


    else:


        lines_split = content.split('\n')


        if len(lines_split) > 1 and lines_split[-1] == '':


            num_lines = len(lines_split) - 1


        else:


            num_lines = len(lines_split)





    # Count words


    words = re.findall(r'[a-zA-Z]+', content)


    words = [w.lower() for w in words]


    total_words = len(words)





    if not words:


        top_word = "-"


        top_count = 0


    else:


        counts = Counter(words)


        # Sort by frequency descending, then by word alphabetically ascending


        sorted_counts = sorted(counts.items(), key=lambda item: (-item[1], item[0]))


        top_word, top_count = sorted_counts[0]





    print(f"lines: {num_lines}")


    print(f"words: {total_words}")


    print(f"top: {top_word} {top_count}")





if __name__ == "__main__":


    main()
```

```python
import sys


import re





def wordstats(filename):


    """Calculates and prints word statistics for a given text file."""





    try:


        with open(filename, 'r') as f:


            text = f.read()


    except FileNotFoundError:


        print(f"Error: File '{filename}' not found.")


        return





    lines = text.splitlines()


    num_lines = len(lines)





    # Clean and tokenize the text


    words = re.findall(r'\b[a-z]+\b', text.lower())


    word_counts = {}


    for word in words:


        word_counts[word] = word_counts.get(word, 0) + 1





    # Find the most frequent word


    if not word_counts:


        top_word = "- 0"


    else:


        top_word = max(word_counts, key=lambda k: (word_counts[k], k))





    print(f"lines: {num_lines}")


    print(f"words: {len(words)}")


    print(f"top: {top_word}")








if __name__ == "__main__":


    if len(sys.argv) != 2:


        print("Usage: python wordstats.py <path-to-text-file>")


    else:


        filename = sys.argv[1]


        wordstats(filename)
```
