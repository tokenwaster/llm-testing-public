# collkit.py
def deduplicate(items):
    """Removes duplicate items from a list while preserving order."""
    seen = set()
    result = []
    for item in items:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

def partition(items, key):
    """Partitions a list into two lists based on a key function."""
    matches = []
    non_matches = []
    for item in items:
        if key(item):
            matches.append(item)
        else:
            non_matches.append(item)
    return matches, non_matches

def sliding_window(items, window_size):
    """Generates a sliding window of a specified size over a list."""
    windows = []
    for i in range(len(items) - window_size + 1):
        windows.append(items[i:i+window_size])
    return windows

# collkit.py
def deduplicate(items):
    """Removes duplicate items from a list while preserving order."""
    seen = set()
    result = []
    for item in items:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

def partition(items, key):
    """Partitions a list into two lists based on a key function."""
    matches = []
    non_matches = []
    for item in items:
        if key(item):
            matches.append(item)
        else:
            non_matches.append(item)
    return matches, non_matches

def sliding_window(items, window_size):
    """Generates a sliding window of a specified size over a list."""
    windows = []
    for i in range(len(items) - window_size + 1):
        windows.append(items[i:i+window_size])
    return windows
